SELECCIÓN Y FILTRADO DE DATOS
-----------------------------------------------------
df['columna']                  # Selección de una columna
-- SQL: SELECT columna FROM tabla;

df[['col1','col2']]            # Selección múltiple
-- SQL: SELECT col1, col2 FROM tabla;

df.loc[5]                      # Fila por índice
-- SQL: SELECT * FROM tabla WHERE id = 5;

df.iloc[0:5]                   # Filas por posición
-- SQL: SELECT * FROM tabla OFFSET 0 LIMIT 5;

df[df['columna'] > 100]        # Filtrar por condición
-- SQL: SELECT * FROM tabla WHERE columna > 100;

df[(df['col1'] > 100) & (df['col2'] == 'A')]
-- SQL: SELECT * FROM tabla WHERE col1 > 100 AND col2 = 'A';

df[df['columna'].isin([1,2,3])]
-- SQL: SELECT * FROM tabla WHERE columna IN (1,2,3);

df[~df['columna'].isin([1,2,3])]
-- SQL: SELECT * FROM tabla WHERE columna NOT IN (1,2,3);

-----------------------------------------------------
EDICIÓN Y TRANSFORMACIÓN DE DATOS
-----------------------------------------------------
# Renombrar columnas
df.rename(columns={'vieja':'nueva'}, inplace=True)
-- SQL: ALTER TABLE tabla RENAME COLUMN vieja TO nueva;

# Crear nueva columna
df['nueva'] = df['col1'] + df['col2']
-- SQL: ALTER TABLE tabla ADD COLUMN nueva tipo;
-- SQL: UPDATE tabla SET nueva = col1 + col2;

# Aplicar función a una columna
df['columna'] = df['columna'].apply(lambda x: x.upper())
-- SQL: UPDATE tabla SET columna = UPPER(columna);

# Cambiar tipo de dato
df['columna'] = df['columna'].astype(float)
-- SQL: ALTER TABLE tabla ALTER COLUMN columna TYPE float;

# Reemplazar valores
df['columna'] = df['columna'].replace({'antiguo':'nuevo'})
-- SQL: UPDATE tabla SET columna = 'nuevo' WHERE columna = 'antiguo';

# Eliminar columnas
df.drop(columns=['columna'], inplace=True)
-- SQL: ALTER TABLE tabla DROP COLUMN columna;

# Eliminar filas por índice
df.drop(index=[0,1], inplace=True)
-- SQL: DELETE FROM tabla WHERE id IN (valor0, valor1);

# Eliminar duplicados
df.drop_duplicates(inplace=True)
-- SQL: DELETE FROM tabla t1 USING tabla t2 WHERE t1.id < t2.id AND t1.campo = t2.campo;

FLUJO ETL COMPLETO (EJEMPLO)
-----------------------------------------------------
import pandas as pd

# 1. EXTRAER
ventas = pd.read_csv("data/ventas.csv")    # SELECT * FROM ventas;
clientes = pd.read_csv("data/clientes.csv")# SELECT * FROM clientes;

# 2. TRANSFORMAR
# Filtrar solo ventas mayores a 1000
ventas = ventas[ventas['monto'] > 1000]    # WHERE monto > 1000

# Unir con clientes
df_final = pd.merge(ventas, clientes, on='cliente_id', how='left') # LEFT JOIN

# Calcular columna nueva
df_final['monto_usd'] = df_final['monto'] / 950  # tasa ficticia

# Limpiar espacios y poner nombres en mayúsculas
df_final['nombre_cliente'] = df_final['nombre_cliente'].str.strip().str.upper()

# Eliminar duplicados
df_final.drop_duplicates(inplace=True)

# 3. CARGAR
df_final.to_csv("data/ventas_procesadas.csv", index=False)
-- SQL equivalente:
-- DELETE FROM ventas_procesadas;
-- INSERT INTO ventas_procesadas SELECT ...;
